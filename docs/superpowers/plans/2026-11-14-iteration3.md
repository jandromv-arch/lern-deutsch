```ts
export function normalizeGerman(s: string): string {
  return s.toLowerCase().trim().replace(/\s+/g, " ")
    .replace(/ä/g, "ae").replace(/ö/g, "oe").replace(/ü/g, "ue").replace(/ß/g, "ss")
    .replace(/[.,!?;:]/g, "");
}
export interface Type { promptFr: string; answer: string }
export function buildType(item: VocabItem): Type { return { promptFr: item.fr, answer: item.de }; }
export function checkTyped(input: string, answer: string): boolean {
  return normalizeGerman(input) === normalizeGerman(answer);
}
```

```tsx
import { useState } from "react";
import type { VocabItem } from "../content/types";
import { buildType, checkTyped } from "../engine/exercises";
import { AudioButton } from "./AudioButton";

export function TypeInput({ item, onResult }: { item: VocabItem; pool?: VocabItem[]; onResult: (c: boolean) => void }) {
  const t = buildType(item);
  const [value, setValue] = useState("");
  const [done, setDone] = useState<null | boolean>(null);
  const submit = () => {
    if (done !== null) return;
    const ok = checkTyped(value, t.answer);
    setDone(ok);
    setTimeout(() => onResult(ok), 600);
  };
  return (
    <div className="p-6 space-y-4 text-center">
      <div className="text-neutral-500 dark:text-neutral-400">{t.promptFr}</div>
      <div className="flex items-center justify-center gap-2">
        <input value={value} onChange={(e) => setValue(e.target.value)}
          onKeyDown={(e) => e.key === "Enter" && submit()}
          autoCapitalize="off" autoCorrect="off" spellCheck={false} lang="de"
          className={`border-b-2 bg-transparent text-xl text-center outline-none px-2 py-1 ${done===null?"border-accent":done?"border-good text-good":"border-bad text-bad"}`}
          placeholder="…" aria-label="Réponse en allemand" />
        <AudioButton text={t.answer} />
      </div>
      {done === false && <div className="text-sm">Réponse : <span className="font-semibold">{t.answer}</span></div>}
      <button onClick={submit} disabled={done!==null}
        className="w-full py-3 rounded-xl bg-accent text-white disabled:opacity-40">Vérifier</button>
    </div>
  );
}
```

```ts
export function validateDialogues(ds: Dialogue[]): string[] {
  const errs: string[] = []; const seen = new Set<string>();
  for (const d of ds) {
    if (seen.has(d.id)) errs.push(`duplicate dialogue id: ${d.id}`); seen.add(d.id);
    if (d.lines.length < 2) errs.push(`dialogue ${d.id} needs >=2 lines`);
    for (const [i, l] of d.lines.entries()) {
      if (!l.de.trim()) errs.push(`${d.id} line ${i} missing de`);
      if (!l.fr.trim()) errs.push(`${d.id} line ${i} missing fr`);
    }
  }
  return errs;
}
```
